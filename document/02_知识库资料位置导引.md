### 3. 匹配度算法核心

**文件**：`src/utils/match.ts`

**关键函数**：
```typescript
export const calcMatchScore = (
  studentDims: Record<string, number>,
  jobDims: Record<string, number>
): number => {
  // 相对缺陷模型
  const dims = ['professional', 'certificate', 'innovation', 'learning', 'stress', 'communication', 'internship']
  
  const deficits = dims.map(d => {
    const studentScore = studentDims[d] || 0
    const jobRequirement = jobDims[d] || 0
    const deficit = Math.max(0, jobRequirement - studentScore)
    // 相对缺陷 = 缺陷 / 岗位要求
    return deficit / Math.max(1, jobRequirement)
  })
  
  const avgDeficit = deficits.reduce((a, b) => a + b) / dims.length
  return Math.max(0, Math.min(100, Math.round(100 * (1 - avgDeficit))))
}

export const getMatchTagType = (score: number): string => {
```

### 3. 关键算法：buildWhereClause

**文件**：`backend/app.js` 中的 `buildWhereClause(query)` 函数

**功能**：构建SQL WHERE条件，支持多维度筛选

**参数处理**：
```javascript
function buildWhereClause(query) {
  const whereParts = []
  const queryParams = []
  
  // 1. 搜索（岗位名或公司名）
  if (query.search) {
    whereParts.push('(job_name LIKE ? OR company LIKE ?)')
    const pattern = `%${query.search}%`
    queryParams.push(pattern, pattern)
  }
  
  // 2. 技能过滤（JSON_CONTAINS）
  if (query.skill) {
    whereParts.push(`JSON_CONTAINS(requirements->'$.professionalSkills', JSON_QUOTE(?))`)
    queryParams.push(query.skill)
  }
  
  // 3. 行业过滤（clean_industry）
  if (query.industry) {
    whereParts.push(`JSON_CONTAINS(clean_industry, JSON_QUOTE(?))`)
    queryParams.push(query.industry)
  }
  
  // 4. 城市过滤
  if (query.city) {
    whereParts.push('location = ?')
    queryParams.push(query.city)
  }
  
  // 5. 薪资范围过滤
  if (query.salary) {
    // 解析 "15k-25k" 格式，转换为数字范围
    const [min, max] = query.salary.split('-').map(s => parseInt(s) * 1000)
    whereParts.push('CAST(salary AS UNSIGNED) BETWEEN ? AND ?')
    queryParams.push(min, max)
  }
  
  return {
    whereParts,
    queryParams,
    sqlPart: whereParts.length > 0 ? 'WHERE ' + whereParts.join(' AND ') : ''
  }
}
```

**使用示例**：
```javascript
// 搜索 "Java" 技能 + "杭州" 城市
const { sqlPart, queryParams } = buildWhereClause({
  skill: 'Java',
  city: '杭州'
})

const sql = `SELECT * FROM job_standard_profile ${sqlPart} LIMIT 10`
const result = await pool.query(sql, queryParams)
```

### 4. 简历解析：Dify Agent集成

**文件**：`backend/app.js` 中的 `POST /api/analyze` 路由

**流程**：
```javascript
app.post('/api/analyze', upload.single('resume'), async (req, res) => {
  try {
    // 1. 提取简历文本
    const file = req.file
    let resumeText = ''
    
    if (file.mimetype === 'application/pdf') {
      // PDF 解析（pdfParse）
      const pdfData = await pdf(file.buffer)
      resumeText = pdfData.text.substring(0, 4000)  // 最多4000字符
    } else if (file.originalname.endsWith('.docx')) {
      // Word 解析（mammoth）
      const result = await mammoth.extractRawText({ buffer: file.buffer })
      resumeText = result.value.substring(0, 4000)
    } else {
      // 纯文本
      resumeText = file.buffer.toString().substring(0, 4000)
    }
    
    // 2. 调用Dify Agent简历解析
    const difyResponse = await axios.post(
      `${DIFY_API_URL}/chat-messages`,
      {
        inputs: { resume_text: resumeText },  // 重要：变量名必须是 resume_text
        response_mode: 'blocking'               // Agent支持blocking模式
      },
      {
        headers: { Authorization: `Bearer ${DIFY_API_KEY_RESUME_ANALYSIS}` }
      }
    )
    
    let answer = difyResponse.data.answer
    
    // 3. 解析JSON结果
    const match = answer.match(/\{[\s\S]*\}/)
    const jsonResult = match ? JSON.parse(match[0]) : null
    
    if (jsonResult) {
      res.json({
        code: 0,
        data: jsonResult,
        source: 'ai'
      })
    } else {
      throw new Error('JSON parse failed')
    }
  } catch (err) {
    // 降级方案：本地规则
    const mockProfile = buildMockProfile(resumeText)
    res.json({
      code: 0,
      data: mockProfile,
      source: 'fallback'
    })
  }
})
```

**关键点**：
- 上传文件必须是PDF/Word/Text
- 从文件中提取文本，截断到4000字符
- 调用Dify Agent，输入参数名必须是`resume_text`
- **必须在Dify中添加`resume_text`字符串变量**，否则会ECONNRESET错误
- 尝试解析JSON响应，失败则用本地规则降级

### 5. AI对话：Dify Agent集成（流式）

**文件**：`backend/app.js` 中的 `POST /api/chat` 路由

**流程**：
```javascript
app.post('/api/chat', upload.single('file'), async (req, res) => {
  try {
    const { message, conversationId } = req.body
    const file = req.file
    
    let messageToSend = message || '请帮我分析这份简历'
    
    // 如果有文件附件，提取文本并追加到消息
    if (file) {
      let fileText = ''
      if (file.mimetype === 'application/pdf') {
        const pdfData = await pdf(file.buffer)
        fileText = pdfData.text.substring(0, 2000)
      } else if (file.originalname.endsWith('.docx')) {
        const result = await mammoth.extractRawText({ buffer: file.buffer })
        fileText = result.value.substring(0, 2000)
      }
      messageToSend = `[附件内容]\n${fileText}\n\n[用户问题]\n${message}`
    }
    
    // 调用Dify Agent（流式模式）
    const axiosResponse = await axios.post(
      `${DIFY_API_URL}/chat-messages`,
      {
        query: messageToSend,
        conversation_id: conversationId || undefined,
        response_mode: 'streaming'  // 关键：Agent必须用streaming
      },
      {
        headers: { Authorization: `Bearer ${DIFY_API_KEY}` },
        responseType: 'stream'
      }
    )
    
    // 解析SSE流
    let fullAnswer = ''
    let newConversationId = conversationId || ''
    
    return new Promise((resolve, reject) => {
      axiosResponse.data.on('data', (chunk) => {
        const lines = chunk.toString().split('\n')
        lines.forEach(line => {
          if (line.startsWith('data:')) {
            try {
              const json = JSON.parse(line.slice(5))
              if (json.event === 'agent_message') {
                fullAnswer += json.answer  // 累加delta
              }
              if (json.conversation_id) {
                newConversationId = json.conversation_id
              }
            } catch (e) {}
          }
        })
      })
      
      axiosResponse.data.on('end', () => {
        res.json({
          code: 0,
          answer: fullAnswer,
          conversationId: newConversationId
        })
        resolve()
      })
      
      axiosResponse.data.on('error', reject)
    })
  } catch (err) {
    res.status(500).json({
      code: 1,
      message: 'AI服务暂时不可用，请稍后重试'
    })
  }
})
```

**关键点**：
- 使用`response_mode: 'streaming'`（Agent only）
- 通过`conversationId`维护多轮对话
- SSE事件：`agent_message` 包含answer delta
- 累加所有delta得到完整回复

```javascript
  const extractSkills = () => {

    },
```

## 关键算法和工具函数

### 1. 相对缺陷匹配算法

**位置**：`src/utils/match.ts`

**算法原理**：
```
传统模型：绝对缺陷 = 岗位要求 - 学生分数
问题：低要求岗位也能达到100分，无法区分

相对缺陷模型：相对缺陷 = (岗位要求 - 学生分数) / 岗位要求
优势：
- 缺陷被相对化
- 低要求岗位不会虚高
- 分布更合理（50-95%）
- 用户易于理解

匹配度 = 100 * (1 - 平均相对缺陷)
```

**伪代码**：
```
for each dimension in [professional, certificate, ...]:
  deficit = max(0, jobRequirement - studentScore)
  relativeDeficit = deficit / max(1, jobRequirement)
  
avgDeficit = sum(relativeDeficits) / 7
matchScore = max(0, min(100, 100 * (1 - avgDeficit)))
```

### 2. 技能过滤（JSON_CONTAINS）

**位置**：`backend/app.js` 的 `buildWhereClause` 函数

**SQL示例**：
```sql
-- 搜索包含 "Java" 技能的所有岗位
SELECT * FROM job_standard_profile
WHERE JSON_CONTAINS(requirements->'$.professionalSkills', JSON_QUOTE('Java'))

-- 搜索行业包含 "互联网" 的岗位
SELECT * FROM job_standard_profile
WHERE JSON_CONTAINS(clean_industry, JSON_QUOTE('互联网'))
```

### 3. 分页查询优化

**位置**：`backend/app.js` 的分页API路由

**实现**：
```javascript
const page = parseInt(req.query.page) || 1
const size = parseInt(req.query.size) || 8
const offset = (page - 1) * size

const { sqlPart, queryParams } = buildWhereClause(req.query)
const sql = `
  SELECT * FROM job_standard_profile
  ${sqlPart}
  LIMIT ? OFFSET ?
`
queryParams.push(size, offset)

const result = await pool.query(sql, queryParams)
const total = await pool.query(`SELECT COUNT(*) as count FROM job_standard_profile ${sqlPart}`, queryParams.slice(0, -2))

res.json({
  code: 0,
  data: result[0],
  page,
  size,
  total: total[0][0].count
})
```
